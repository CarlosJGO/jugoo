## Dependencias del sistema para Jugoo (Arch / CachyOS)
## Instálalas con: python3 acomodador.py
## o a mano con las líneas `sudo pacman -S` de abajo.

# Runtime GTK / Python
sudo pacman -S python-gobject python-cairo gtk3 gtk-layer-shell wtype

# Polkit authentication agent (obligatorio en Hyprland para pkexec / SDDM).
# Sin esto, pkexec falla con «Not authorized» / journal «No session for cookie»
# aunque el usuario esté en wheel y sudo funcione.
sudo pacman -S hyprpolkitagent
# systemctl --user enable --now hyprpolkitagent.service
# (acomodador.py lo activa; o en hyprland: exec-once = systemctl --user start hyprpolkitagent)

# Bluetooth (BlueZ): Jugoo habla con bluetoothd vía D-Bus (org.bluez).
# Transferencia de archivos: Object Push vía org.bluez.obex (obexd, bus de sesión).
# No se empaqueta BlueZ en el repo; debe venir del sistema.
# Comprobar (sin alterar el servicio): `systemctl is-active bluetooth`, `bluetoothctl show`
sudo pacman -S bluez bluez-utils bluez-obex
# systemctl enable --now bluetooth.service   # solo si quieres activarlo a mano
# systemctl --user enable --now obex.service # transferencia de archivos (enviar/recibir)
